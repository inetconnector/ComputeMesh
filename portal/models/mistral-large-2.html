Drop-in Python OpenAI SDK Integration

from openai import OpenAI

client = OpenAI(
    base_url="https://inetconnector.com/v1",
    api_key="cm_live_YOUR_API_KEY",
)

resp = client.chat.completions.create(
    model="mistral-large:latest",
    messages=[{"role": "user", "content": "Summarize key architecture trade-offs."}],
)
print(resp.choices[0].message.content)
